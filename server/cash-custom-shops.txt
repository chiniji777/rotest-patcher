function	script	F_RT_OpenBox	{
	.@box=getarg(0);.@item=getarg(1);.@amount=getarg(2);
	if (countitem(.@box)<1) return;
	if (countitem(.@item)>0) {dispbottom "Store existing contents of this type in Kafra first. Box was not consumed.";return;}
	if (!checkweight(.@item,.@amount)) {dispbottom "Not enough inventory space or weight. Box was not consumed.";return;}
	delitem .@box,1;
	getitem .@item,.@amount;
	return;
}
function	script	F_RT_PvpPotion	{
	.@item=getarg(0);
	if (countitem(.@item)<1) return;
	if (!getmapflag(strcharinfo(3),mf_pvp)) {dispbottom "PVP-only potion. Item was not consumed.";return;}
	delitem .@item,1;
	heal getarg(1),getarg(2);
	return;
}
